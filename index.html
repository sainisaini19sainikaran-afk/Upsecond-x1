let speakSessionId = 0; let isSpeakingActive = false; const SETTINGS_KEY = 'upsecond_voice_settings_v2'; function saveVoiceSettings() { const data = { lang: (document.getElementById('langFilter') || {}).value || 'ALL', gender: (document.getElementById('genderFilter') || {}).value || 'FEMALE', style: (document.getElementById('styleCommand') || {}).value || 'natural', rate: (document.getElementById('rateRange') || {}).value || '0.98', pitch: (document.getElementById('pitchRange') || {}).value || '1.10' }; try { localStorage.setItem(SETTINGS_KEY, JSON.stringify(data)); } catch (e) {} } function loadVoiceSettings() { try { const raw = localStorage.getItem(SETTINGS_KEY); if (!raw) return; const d = JSON.parse(raw); if (d.lang && document.getElementById('langFilter')) document.getElementById('langFilter').value = d.lang; if (d.gender && document.getElementById('genderFilter')) document.getElementById('genderFilter').value = d.gender; if (d.style && document.getElementById('styleCommand')) document.getElementById('styleCommand').value = d.style; if (d.rate && document.getElementById('rateRange')) document.getElementById('rateRange').value = d.rate; if (d.pitch && document.getElementById('pitchRange')) document.getElementById('pitchRange').value = d.pitch; } catch (e) {} } function splitIntoNaturalChunks(text) { const cleaned = text.replace(/\s+/g, ' ').trim(); if (!cleaned) return []; const rawParts = cleaned.match(/[^.!?।]+[.!?।]*|[^.!?।]+$/g) || [cleaned]; const chunks = []; rawParts.forEach(part => { const p = part.trim(); if (!p) return; if (p.length <= 180) { chunks.push(p); } else { const subParts = p.split(/,|;|\n/).map(s => s.trim()).filter(Boolean); subParts.forEach(sp => { if (sp.length <= 180) { chunks.push(sp); } else { for (let i = 0; i < sp.length; i += 170) { chunks.push(sp.slice(i, i + 170).trim()); } } }); } }); return chunks.length ? chunks : [cleaned]; } function pickBestVoice(langMode, genderMode) { const voices = allVoices || []; const hiEn = voices.filter(isHindiOrEnglish); const langPool = hiEn.filter(v => { const l = (v.lang || '').toLowerCase(); if (langMode === 'HI') return l.startsWith('hi'); if (langMode === 'EN') return l.startsWith('en'); return true; }); const targetPool = langPool.length ? langPool : (hiEn.length ? hiEn : voices); const exactGender = targetPool.find(v => detectGender(v) === genderMode); if (exactGender) return exactGender; const anyGenderInAll = voices.find(v => detectGender(v) === genderMode); if (anyGenderInAll) return anyGenderInAll; const nonOpposite = targetPool.find(v => detectGender(v) !== (genderMode === 'FEMALE' ? 'MALE' : 'FEMALE')); return nonOpposite || targetPool[0] || null; } function computeCommandParams(genderMode, styleMode, userRate, userPitch) { let rate = Number(userRate) || (genderMode === 'MALE' ? 0.94 : 0.98); let pitch = Number(userPitch) || (genderMode === 'MALE' ? 0.84 : 1.12); if (styleMode === 'expressive') { rate = Math.min(1.3, rate + 0.06); pitch = genderMode === 'MALE' ? Math.min(0.95, pitch + 0.04) : Math.min(1.28, pitch + 0.06); } else if (styleMode === 'calm') { rate = Math.max(0.72, rate - 0.08); pitch = genderMode === 'MALE' ? Math.max(0.68, pitch - 0.04) : Math.max(0.88, pitch - 0.04); } if (genderMode === 'MALE' && pitch > 0.96) { pitch = 0.84; } if (genderMode === 'FEMALE' && pitch < 0.92) { pitch = 1.12; } return { rate: Number(Math.max(0.65, Math.min(1.4, rate)).toFixed(2)), pitch: Number(Math.max(0.65, Math.min(1.35, pitch)).toFixed(2)) }; } function applyCommandPreset() { const gender = (document.getElementById('genderFilter') || {}).value || 'FEMALE'; const style = (document.getElementById('styleCommand') || {}).value || 'natural'; const rateEl = document.getElementById('rateRange'); const pitchEl = document.getElementById('pitchRange'); const defaultRate = gender === 'MALE' ? 0.94 : 0.98; const defaultPitch = gender === 'MALE' ? 0.84 : 1.12; const tuned = computeCommandParams(gender, style, defaultRate, defaultPitch); if (rateEl) rateEl.value = String(tuned.rate); if (pitchEl) pitchEl.value = String(tuned.pitch); updateSliders(); populateVoices(); saveVoiceSettings(); } function speakText() { const statusEl = document.getElementById('voiceStatus'); const textEl = document.getElementById('voiceText'); const fullText = textEl ? voiceSelect.value : 'AUTO'; const langMode = (document.getElementById('langFilter') || {}).value || 'ALL'; const genderMode = (document.getElementById('genderFilter') || {}).value || 'FEMALE'; const styleMode = (document.getElementById('styleCommand') || {}).value || 'natural'; const rateEl = document.getElementById('rateRange'); const pitchEl = document.getElementById('pitchRange'); const chosenVoice = (selectedVal !== 'AUTO' && selectedVal !== '' && allVoices[Number(selectedVal)]) ? 'hi-IN' : 'en-IN'; } utter.rate = params.rate; utter.pitch = params.pitch; utter.onstart = () => { if (currentSession !== speakSessionId) return; setSpeakingLine(true); if (statusEl) { statusEl.textContent = `Speaking ${genderMode === 'MALE' ?
